using UnityEngine;

// Assignment 03 Part A: a second physics force, separate from knockback.
[RequireComponent(typeof(Collider))]
public class BouncePad : MonoBehaviour
{
    [SerializeField] float bounceStrength = 8f;

    void OnTriggerEnter(Collider other)
    {
        if (!other.CompareTag("Player")) return;
        Rigidbody body = other.attachedRigidbody;
        if (body == null) return;

        // Cancel downward falling speed so this pad always gives a visible bounce.
        Vector3 v = body.linearVelocity;
        if (v.y < 0f) body.linearVelocity = new Vector3(v.x, 0f, v.z);
        body.AddForce(Vector3.up * bounceStrength, ForceMode.VelocityChange);
    }
}

using UnityEngine;

public class CameraFollow : MonoBehaviour
{
    [SerializeField] private PlayerController target;
    [SerializeField] private Vector3 offset = new Vector3(0f, 4f, -7f);
    [SerializeField, Min(0.01f)] private float sharpness = 5f;
    [SerializeField, Min(0f)] private float lookAhead = 0.35f;
    [SerializeField] private float lookHeight = 1f;

    private void Awake()
    {
        if (target == null)
            target = FindAnyObjectByType<PlayerController>();
    }

    private void LateUpdate()
    {
        if (target == null)
        {
            target = FindAnyObjectByType<PlayerController>();
            if (target == null) return;
        }

        Rigidbody body = target.GetComponent<Rigidbody>();
        if (body == null) return;

#if UNITY_6000_0_OR_NEWER
        Vector3 flatVelocity = body.linearVelocity;
#else
        Vector3 flatVelocity = body.velocity;
#endif
        flatVelocity.y = 0f;

        Vector3 aim = target.transform.position
                      + Vector3.up * lookHeight
                      + flatVelocity * lookAhead;
        Vector3 wanted = aim + offset;

        float blend = 1f - Mathf.Exp(-sharpness * Time.deltaTime);
        transform.position = Vector3.Lerp(transform.position, wanted, blend);
        transform.LookAt(aim);
    }
}

public static class Ease
{
    public static float OutBack(float t)
    {
        const float c1 = 1.70158f;
        const float c3 = c1 + 1f;
        float u = t - 1f;
        return 1f + c3 * u * u * u + c1 * u * u;
    }
}

using UnityEngine;

// Lecture 07's patrol modes, expanded to the assignment's 3 states.
public enum PatrolMode { Loop, PingPong, Random }
public enum EnemyState { Patrol, Chase, Return }

[RequireComponent(typeof(Rigidbody))]
public class EnemyPatrol : MonoBehaviour
{
    [SerializeField] Transform[] points;
    [SerializeField] PatrolMode mode = PatrolMode.Loop;
    [SerializeField] float speed = 2f;
    [SerializeField] float waitTime = 0f;
    [SerializeField] float turnSpeed = 360f;
    [SerializeField] float arriveDistance = 0.2f;

    [Header("Chase and return")]
    [SerializeField] float chaseSpeed = 3.5f;
    [SerializeField] float chaseRange = 4f;
    [SerializeField] float giveUpRange = 6f;

    Transform player;
    EnemyState state = EnemyState.Patrol;
    int index, step = 1;
    float waitUntil;

    void Awake()
    {
        Rigidbody body = GetComponent<Rigidbody>();
        body.isKinematic = true;
        body.useGravity = false;
    }

    void Start()
    {
        PlayerController found = FindAnyObjectByType<PlayerController>();
        if (found != null) player = found.transform;
        if (giveUpRange <= chaseRange) giveUpRange = chaseRange + 2f;
    }

    void Update()
    {
        if (points == null || points.Length < 3 || player == null) return;

        float distance = Vector3.Distance(transform.position, player.position);
        if (state != EnemyState.Chase && distance < chaseRange)
            state = EnemyState.Chase;
        else if (state == EnemyState.Chase && distance > giveUpRange)
            state = EnemyState.Return;

        if (state == EnemyState.Chase)
        {
            MoveTo(player.position, chaseSpeed);
        }
        else if (state == EnemyState.Return)
        {
            MoveTo(points[index].position, speed);
            if (Arrived(points[index].position)) state = EnemyState.Patrol;
        }
        else if (Time.time >= waitUntil)
        {
            MoveTo(points[index].position, speed);
            if (Arrived(points[index].position))
            {
                index = NextIndex();
                waitUntil = Time.time + waitTime;
            }
        }
    }

    bool Arrived(Vector3 target)
    {
        // Assignment 03: distance tolerance, not position == target.
        target.y = transform.position.y;
        return Vector3.Distance(transform.position, target) < arriveDistance;
    }

    void MoveTo(Vector3 target, float moveSpeed)
    {
        // Keep the guard on its own height; use open patrol routes (no wall pathfinding).
        target.y = transform.position.y;
        Vector3 direction = target - transform.position;
        transform.position = Vector3.MoveTowards(
            transform.position, target, moveSpeed * Time.deltaTime);

        if (direction.sqrMagnitude > 0.001f)
        {
            Quaternion look = Quaternion.LookRotation(direction);
            transform.rotation = Quaternion.RotateTowards(
                transform.rotation, look, turnSpeed * Time.deltaTime);
        }
    }

    int NextIndex()
    {
        if (mode == PatrolMode.Loop) return (index + 1) % points.Length;
        if (mode == PatrolMode.PingPong)
        {
            if (index + step < 0 || index + step >= points.Length) step = -step;
            return index + step;
        }
        int next = Random.Range(0, points.Length);
        return next == index ? (next + 1) % points.Length : next;
    }
}

using System.Collections;
using UnityEngine;

// The same script works on spikes, guards and a timed crusher (Lecture 07).
[RequireComponent(typeof(Collider))]
public class Hazard : MonoBehaviour
{
    [SerializeField] int damage = 1;
    [SerializeField] float knockback = 6f;
    [SerializeField] float knockTime = 0.25f;

    [Header("Timed crusher (restTime = 0 for spikes and guards)")]
    [SerializeField] float restTime = 0f;
    [SerializeField] float warnTime = 0.7f;
    [SerializeField] float strikeTime = 0.4f;
    [SerializeField] GameObject warning;
    [SerializeField] GameObject strikeVisual;

    IEnumerator Start()
    {
        if (restTime <= 0f) yield break;

        Collider zone = GetComponent<Collider>();
        if (warning != null) warning.SetActive(false);
        if (strikeVisual != null) strikeVisual.SetActive(false);

        while (true)
        {
            zone.enabled = false;
            yield return new WaitForSeconds(restTime);

            // The assignment requires a visible warning of at least 0.5 s.
            if (warning != null) warning.SetActive(true);
            yield return new WaitForSeconds(Mathf.Max(0.5f, warnTime));

            if (warning != null) warning.SetActive(false);
            if (strikeVisual != null) strikeVisual.SetActive(true);
            zone.enabled = true;
            yield return new WaitForSeconds(strikeTime);

            if (strikeVisual != null) strikeVisual.SetActive(false);
        }
    }

    void OnTriggerStay(Collider other)
    {
        Rigidbody body = other.attachedRigidbody;
        if (body == null || !body.CompareTag("Player")) return;

        Health health = body.GetComponent<Health>();
        if (health == null || !health.TakeDamage(damage)) return;

        Vector3 away = body.position - transform.position;
        away.y = 0f;
        if (away.sqrMagnitude < 0.001f)
        {
            away = body.transform.forward;
            away.y = 0f;
            if (away.sqrMagnitude < 0.001f) away = Vector3.forward;
        }

        PlayerController controller = body.GetComponent<PlayerController>();
        if (controller != null)
            controller.Knockback(away.normalized * knockback, knockTime);
    }
}

using UnityEngine;

// Lecture 07 hearts, i-frames and optional one-use shield.
public class Health : MonoBehaviour
{
    [SerializeField] int max = 3;
    [SerializeField] float iFrames = 1f;
    public bool shielded;

    int current;
    float invulnerableUntil;

    public int Current => current;
    public int Max => max;

    void Awake() => current = max;

    // True ONLY when damage landed; Hazard then applies knockback.
    public bool TakeDamage(int amount)
    {
        if (amount <= 0 || current <= 0 || Time.time < invulnerableUntil)
            return false;

        invulnerableUntil = Time.time + iFrames;
        if (shielded)
        {
            shielded = false;
            return false;
        }

        current = Mathf.Max(0, current - amount);
        if (GameManager.Instance != null)
        {
            GameManager.Instance.UpdateUI();
            if (current <= 0) GameManager.Instance.PlayerDied();
        }
        return true;
    }

    public void Restore(float protection)
    {
        current = max;
        shielded = false;
        invulnerableUntil = Time.time + protection;
        if (GameManager.Instance != null) GameManager.Instance.UpdateUI();
    }
}


##### Assets/Scipts/PlayerController.cs
using System.Collections;
using UnityEngine;


[RequireComponent(typeof(Rigidbody))]
public class PlayerController : MonoBehaviour
{
    [SerializeField] float speed = 5f;

    Rigidbody rb;
    Vector3 move;
    float speedMultiplier = 1f;
    float noControlUntil;
    Coroutine boost;

    void Awake() => rb = GetComponent<Rigidbody>();

    void Update()
    {
        float h = Input.GetAxis("Horizontal");
        float v = Input.GetAxis("Vertical");
        move = new Vector3(h, 0f, v);
    }

    void FixedUpdate()
    {

        Vector3 direction = Vector3.ClampMagnitude(move, 1f)
            * speed * speedMultiplier;

        if (Time.time >= noControlUntil)
            rb.linearVelocity = new Vector3(
                direction.x, rb.linearVelocity.y, direction.z);
    }

    // Called only when a hazard has actually damaged the player.
    public void Knockback(Vector3 push, float seconds)
    {
        Vector3 v = rb.linearVelocity;
        rb.linearVelocity = new Vector3(0f, v.y, 0f);
        rb.AddForce(push, ForceMode.Impulse);
        noControlUntil = Time.time + seconds;
    }

    // Clear movement before/after respawning.
    public void StopMovement(float seconds = 0f)
    {
        move = Vector3.zero;
        rb.linearVelocity = Vector3.zero;
        rb.angularVelocity = Vector3.zero;
        noControlUntil = Time.time + seconds;
    }


    public void BoostSpeed(float multiplier, float seconds)
    {
        if (boost != null) StopCoroutine(boost);
        boost = StartCoroutine(Boost(multiplier, seconds));
    }

    IEnumerator Boost(float multiplier, float seconds)
    {
        speedMultiplier = multiplier;
        yield return new WaitForSeconds(seconds);
        speedMultiplier = 1f;
        boost = null;
    }
}

        }
        else
        {
            Health health = other.GetComponent<Health>();
            if (health == null) return;
            health.shielded = true;
        }

        Destroy(gameObject);
    }
}

using UnityEngine;

public class Spawner : MonoBehaviour
{
    [SerializeField] GameObject coinPrefab;
    [SerializeField] Transform player;
    [SerializeField] float interval = 1.5f;
    [SerializeField] int maxCoins = 5;
    [SerializeField] float area = 7f;

    void Start()
    {
        if (coinPrefab == null || player == null)
        {
            Debug.LogError("Spawner needs a Coin Prefab and Player reference.", this);
            return;
        }

        InvokeRepeating(nameof(SpawnCoin), 1f, Mathf.Max(0.1f, interval));
    }

    void SpawnCoin()
    {
        if (transform.childCount >= maxCoins)
            return;

        float x = Random.Range(-area, area);
        float z = Random.Range(-area, area);
        Vector3 pos = transform.position + new Vector3(x, 0.5f, z);

        if (Vector3.Distance(pos, player.position) < 2f)
            return;

        Instantiate(coinPrefab, pos, Quaternion.identity, transform);
    }
}
